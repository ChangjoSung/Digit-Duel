'use strict';
// #260 전적 결과 아웃박스 — 서버가 경기 결과를 확정한 순간 **DB 보다 먼저** 로컬 파일에 적어 둔다(JSONL, 경기당 한 줄 = 좌석 행 배열).
// DB 에 들어간 것이 확인된 항목만 지운다. 그래서 DB 장애·프로세스 재시작·크래시로 결과를 잃지 않는다(같은 디스크가 남는 한).
//
// 계약:
//   - add(rows): 한 줄 append + fsync 가 끝난 뒤에만 반환한다. 파일 쓰기가 실패하면 throw — 호출자가 로그하고 메모리에라도 들고 있는다.
//   - 제거는 남은 항목 전체를 임시 파일에 쓰고 fsync 뒤 rename(원자 교체) — 중간에 죽어도 옛 파일 또는 새 파일 하나다.
//   - 기동 시 파일을 다시 읽는다. 깨진 줄(쓰는 중 크래시)은 버리지 않고 `<파일>.bad` 에 그대로 옮겨 적고 로그한다.
//   - 항목은 절대 버리지 않는다. 상한(max)은 "새 경기 시작을 막는" 신호일 뿐이다(full()).
// 모든 파일 작업은 동기다 — 이벤트 루프 한 턴 안에서 append 와 rewrite 가 섞이지 않는다.
// ponytail: 한 줄마다 fsync · 제거마다 전체 재작성 — 경기 종료 빈도(분 단위)에서는 충분하다. 초당 수십 경기면 배치 커밋으로.
const fs = require('fs');
const path = require('path');

function fsyncWrite(file, text, flag) {
  const fd = fs.openSync(file, flag);
  try { fs.writeSync(fd, text); fs.fsyncSync(fd); } finally { fs.closeSync(fd); }
}

// file = null → 메모리 전용(단위 테스트용). 운영 서버는 항상 파일을 준다(server.js).
function createOutbox(file, max) {
  const cap = max > 0 ? max : 200;
  const entries = [];
  if (file) {
    fs.mkdirSync(path.dirname(file), { recursive: true });
    if (fs.existsSync(file)) {
      const bad = [];
      for (const line of fs.readFileSync(file, 'utf8').split('\n')) {
        if (!line.trim()) continue;
        try { const rows = JSON.parse(line); if (Array.isArray(rows) && rows.length) entries.push(rows); else bad.push(line); }
        catch (e) { bad.push(line); }
      }
      if (bad.length) {
        fsyncWrite(file + '.bad', bad.join('\n') + '\n', 'a');
        console.error(`[전적] 아웃박스에 읽을 수 없는 줄 ${bad.length}개 — ${file}.bad 에 보존했다(수동 확인 필요)`);
      }
      rewrite(); // 깨진 줄을 뺀 나머지로 정리(보존은 위에서 끝났다)
    } else {
      fsyncWrite(file, '', 'a'); // 기동 때 쓸 수 있는지 확인한다 — 못 쓰면 여기서 throw(fail-closed)
    }
  }
  function rewrite() {
    if (!file) return;
    const tmp = file + '.tmp';
    fsyncWrite(tmp, entries.map((rows) => JSON.stringify(rows) + '\n').join(''), 'w');
    fs.renameSync(tmp, file);
  }
  return {
    file,
    entries, // 읽기 전용으로 쓴다(순서 = 추가 순서)
    add(rows) {
      if (file) fsyncWrite(file, JSON.stringify(rows) + '\n', 'a');
      entries.push(rows);
    },
    remove(rows) {
      const i = entries.indexOf(rows);
      if (i < 0) return;
      entries.splice(i, 1);
      rewrite();
    },
    full: () => entries.length >= cap,
    size: () => entries.length,
  };
}

module.exports = { createOutbox };
